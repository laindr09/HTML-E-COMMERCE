      }
    };

    const peso = n => "₱" + n.toLocaleString("en-PH", {minimumFractionDigits: 2});
    let current = null, qty = 1;

    function closeAll() {
      document.getElementById("infoOverlay").classList.remove("active");
      document.getElementById("slipOverlay").classList.remove("active");
    }

    // Product info popup
    function openInfo(id) {
      const p = products[id];
      let html = `<h2>${p.title}</h2><p style="font-weight:700;font-size:16px;color:#734F96">${peso(p.price)}</p><p>${p.description}</p>`;
      if (p.included) html += `<h4>What's included</h4><ul>${p.included.map(i => `<li>${i}</li>`).join("")}</ul>`;
      html += `<h4>Best time to use</h4><p>${p.bestTime}</p>`;
      html += `<h4>How to use</h4><ol>${p.steps.map(s => `<li>${s}</li>`).join("")}</ol>`;
      html += `<button class="buy-btn" onclick="addToCart('${id}')">ADD TO CART</button>`;
      document.getElementById("infoContent").innerHTML = html;
      document.getElementById("infoOverlay").classList.add("active");
    }

    // Order slip
    function addToCart(id) {
      current = id; qty = 1;
      document.getElementById("infoOverlay").classList.remove("active");
      renderSlip();
      document.getElementById("slipOverlay").classList.add("active");
    }

    function changeQty(n) {
      qty = Math.max(1, qty + n);
      renderSlip();
    }

    function renderSlip() {
      const p = products[current];
      document.getElementById("slipContent").innerHTML = `
        <h2>Order Slip</h2>
        <div class="slip-row"><span>${p.title}</span><span>${peso(p.price)}</span></div>
        <div class="slip-row"><span>Quantity</span>
          <span class="qty"><button onclick="changeQty(-1)">−</button><span>${qty}</span><button onclick="changeQty(1)">+</button></span>
        </div>
        <div class="slip-row total"><span>Total</span><span>${peso(p.price * qty)}</span></div>
        <button class="buy-btn" onclick="buyNow()">BUY</button>`;
    }

    function buyNow() {
      const p = products[current];
      document.getElementById("slipContent").innerHTML = `
        <h2>Thank you!</h2>
        <p>Your order for ${qty} × ${p.title} (${peso(p.price * qty)}) has been placed.</p>
        <p>Glow. Nurture. Bloom.</p>`;
    }
